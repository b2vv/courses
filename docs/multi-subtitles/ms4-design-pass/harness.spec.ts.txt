async function measure(page: Page) {
  return page.evaluate(() => {
    const vw = document.documentElement.clientWidth;
    const roots = [...document.querySelectorAll('section.lesson-player, section.transcript, .lesson-player-settings')];
    const visible = (e: Element) => {
      const r = e.getBoundingClientRect(), s = getComputedStyle(e);
      return r.width > 0 && r.height > 0 && s.visibility !== 'hidden' && s.clipPath !== 'inset(50%)';
    };
    const all = [...new Set(roots.flatMap(r => [r, ...r.querySelectorAll('*')]))].filter(visible);
    const rect = (e: Element) => { const r = e.getBoundingClientRect(); return { x: r.x, y: r.y, width: r.width, height: r.height, right: r.right, bottom: r.bottom }; };
    const name = (e: Element) => e.getAttribute('aria-label') ?? e.textContent?.trim().slice(0, 100) ?? '';
    const outside = all.filter(e => {
      const r = e.getBoundingClientRect();
      if (r.right <= vw + 0.5 && r.left >= -0.5) return false;
      for (let p = e.parentElement; p; p = p.parentElement) {
        if (['auto', 'scroll'].includes(getComputedStyle(p).overflowX)) return false;
      }
      return true;
    }).map(e => ({ tag: e.tagName, class: String(e.className).slice(0, 90), text: name(e), ...rect(e) }));
    const targets = all.filter(e => e.matches('.transcript > button, .lesson-player-settings label.ui-choice, .lesson-player-settings label.ui-switch, button[aria-label="Налаштування"]'))
      .map(e => ({ name: name(e), ...rect(e), min44: e.getBoundingClientRect().width >= 44 && e.getBoundingClientRect().height >= 44 }));
    // Selected new text and comparison text; measurements do not infer glyph ink clipping.
    const texts = all.filter(e => e.matches('.transcript-empty, .transcript > button, .lesson-player-settings-panel > p, .transcript-text, .lesson-player-settings-panel label.ui-choice'));
    const clipping = texts.map(e => ({ text: name(e), tag: e.tagName, scrollWidth: e.scrollWidth, clientWidth: e.clientWidth, scrollHeight: e.scrollHeight, clientHeight: e.clientHeight, clippedX: e.scrollWidth > e.clientWidth, ...rect(e) }));
    const rgb = (s: string) => {
      const m = s.match(/^rgba?\(([^)]+)\)$/);
      if (!m) return null;
      const v = m[1].split(/[ ,/]+/).filter(Boolean).map(Number);
      return v.length >= 3 && v.every(Number.isFinite) ? [v[0], v[1], v[2], v[3] ?? 1] : null;
    };
    const lum = (v: number[]) => v.slice(0,3).map(c => c / 255).map(c => c <= .04045 ? c / 12.92 : ((c + .055) / 1.055) ** 2.4).reduce((n,c,i) => n + c * [.2126,.7152,.0722][i], 0);
    const contrast = texts.map(e => {
      const cs = getComputedStyle(e), fg = rgb(cs.color);
      let bg: number[] | null = null, bgCSS = '', reason = '', backgroundElement = '';
      const er = e.getBoundingClientRect();
      const textRects: DOMRect[] = [];
      const walker = document.createTreeWalker(e, NodeFilter.SHOW_TEXT);
      for (let node = walker.nextNode(); node; node = walker.nextNode()) {
        if (!node.textContent?.trim()) continue;
        const range = document.createRange(); range.selectNodeContents(node);
        textRects.push(...Array.from(range.getClientRects()).filter(r => r.width > 0 && r.height > 0));
      }
      if (!textRects.length) reason = 'no rendered text rectangles';
      for (let p: Element | null = e; p; p = p.parentElement) {
        const s = getComputedStyle(p);
        if (Number(s.opacity) !== 1 || s.filter !== 'none' || s.backdropFilter !== 'none' || s.mixBlendMode !== 'normal') { reason = 'ancestor opacity/filter/compositing unsupported'; break; }
        if (!bg) {
          if (s.backgroundImage !== 'none') { reason = 'background image/gradient unsupported'; break; }
          const b = rgb(s.backgroundColor);
          if (!b) { reason = `unsupported background colour ${s.backgroundColor}`; break; }
          if (b[3] > 0 && b[3] < 1) { reason = 'translucent background unsupported'; break; }
          if (b[3] === 1) {
            const r = p.getBoundingClientRect();
            if (textRects.some(t => t.left < r.left - .5 || t.right > r.right + .5 || t.top < r.top - .5 || t.bottom > r.bottom + .5)) { reason = 'text outside solid background bounds'; break; }
            bg = b; bgCSS = s.backgroundColor; backgroundElement = `${p.tagName}.${p.className}`;
          }
        }
      }
      const popover = document.querySelector('.lesson-player-settings');
      if (popover && !popover.contains(e)) { const pr = popover.getBoundingClientRect(); if (textRects.some(t => t.left < pr.right && t.right > pr.left && t.top < pr.bottom && t.bottom > pr.top)) reason ||= 'text box overlaps open popover (paint obscured)'; }
      if (!fg || fg[3] !== 1) reason ||= 'foreground not opaque sRGB';
      if (!bg) reason ||= 'no opaque solid background found';
      const ratio = !reason && fg && bg ? (Math.max(lum(fg),lum(bg))+.05)/(Math.min(lum(fg),lum(bg))+.05) : null;
      return { text: name(e), foreground: cs.color, background: bgCSS, backgroundElement, fontSize: cs.fontSize, textRects: textRects.map(t => ({x:t.x,y:t.y,width:t.width,height:t.height})), ratio, status: ratio === null ? 'NOT_TESTED' : ratio >= 4.5 ? 'PASS' : 'FAIL', reason };
    });
    const live = [...document.querySelectorAll('section.transcript [role="status"], section.transcript [role="alert"]')].map(e => ({ role: e.getAttribute('role'), text: name(e) }));
    const track = document.querySelector('video track');
    const video = document.querySelector('video');
    return { viewport: { width: innerWidth, height: innerHeight, clientWidth: vw }, htmlFontSize: getComputedStyle(document.documentElement).fontSize,
      documentScrollWidth: document.documentElement.scrollWidth, overflowPASS: document.documentElement.scrollWidth <= vw && outside.length === 0, outside,
      roots: roots.map(e => ({ selector: `${e.tagName}.${e.className}`, ...rect(e) })), targets, clipping, live, contrast,
      selection: document.querySelector('section.lesson-player')?.getAttribute('data-caption-selection'),
      caption: track instanceof HTMLTrackElement ? { language: track.srclang, mode: track.track.mode, cues: track.track.cues?.length } : null,
      video: video instanceof HTMLVideoElement ? { paused: video.paused, time: video.currentTime, readyState: video.readyState } : null,
      theme: { dataTheme: document.documentElement.getAttribute('data-theme'), colorScheme: getComputedStyle(document.documentElement).colorScheme },
    };
  });
}
async function capture(page: Page, state: string, vp: typeof viewports[number]) {
  for (const theme of ['light', 'dark'] as const) {
    await page.emulateMedia({ colorScheme: theme, reducedMotion: 'reduce' });
    await page.waitForTimeout(300);
    const file = `${state}-${vp.name}-${theme}.png`;
    await page.screenshot({ path: path.join(evidence, file), fullPage: true });
    measurements.push({ state, viewportName: vp.name, themeName: theme, file, ...(await measure(page)) });
  }
  await page.emulateMedia({ colorScheme: 'light' });
  await flush();
  await log(`FACT: Captured ${state} × ${vp.name} light/dark screenshots and geometry/targets/clipping/live/contrast measurements (temporary evidence/${state}-${vp.name}-*.png).`);
}
async function focusState(page: Page) {
  return page.evaluate(() => {
    const e = document.activeElement as HTMLElement | null;
    if (!e) return null;
    const s = getComputedStyle(e), r = e.getBoundingClientRect();
    return { tag: e.tagName, role: e.getAttribute('role'), name: e.getAttribute('aria-label') ?? e.textContent?.trim().slice(0, 90), focusVisible: e.matches(':focus-visible'), outlineWidth: s.outlineWidth, outlineStyle: s.outlineStyle, outlineColor: s.outlineColor, outlineOffset: s.outlineOffset, boxShadow: s.boxShadow, rect: { x:r.x, y:r.y, width:r.width, height:r.height } };
  });
}
async function choose(page: Page, label: string) {
  await gear(page).click();
  await page.getByRole('radiogroup', { name: 'Аудіо', exact: true }).locator('label', { hasText: label }).click();
  await page.keyboard.press('Escape');
}
async function open(browser: Browser, baseURL: string, vp: typeof viewports[number]) {
  // Isolate resume progress between contexts/viewport observations, exactly this lesson only.
  await execute('docker', ['compose','exec','-T','postgres','sh','-eu','-c',
    `exec psql -At -v ON_ERROR_STOP=1 -U "$POSTGRES_USER" -d "$POSTGRES_DB" -c "DELETE FROM lesson_progress WHERE lesson_id = '${lesson}'"`], {cwd:repo,timeout:60_000});

  const context = await browser.newContext({ baseURL, storageState: authStatePath('student'), locale: 'uk-UA', viewport: { width: vp.width, height: vp.height }, colorScheme: 'light', reducedMotion: 'reduce' });
  const page = await context.newPage();
  page.on('request', r => { const u = new URL(r.url()); if (u.pathname.endsWith('/transcript')) requests.push({ viewport: vp.name, kind:'request', pathname:u.pathname, language:u.searchParams.get('language'), at: Date.now() }); });
  page.on('response', r => { const u = new URL(r.url()); if (u.pathname.endsWith('/transcript')) requests.push({ viewport: vp.name, kind:'response', pathname:u.pathname, language:u.searchParams.get('language'), status:r.status(), at: Date.now() }); });
  await page.addInitScript(() => localStorage.setItem('courses.player.prefs.v1', JSON.stringify({ volume:1, muted:false, rate:1, captionsOn:true, autoplayNext:false, audioLanguage:'en' })));
  await page.goto(`/learn/${ns}/${lesson}`);
  await page.evaluate(scale => document.documentElement.style.fontSize = scale === 1 ? '' : `${scale * 100}%`, vp.text);
  await page.getByRole('button', { name:'Відтворити', exact:true }).click();
  await expect(player(page)).toHaveAttribute('data-caption-selection','confirmed:en',{timeout:60_000});
  await expect(panel(page).getByText('English caption 440', {exact:true})).toBeVisible();
  await expect.poll(() => page.locator('video track').evaluate((e: HTMLTrackElement) => e.track.mode)).toBe('showing');
  await log(`FACT: Open ${vp.name}: reset only own lesson progress; confirmed en with showing captions at ${await page.locator('video').evaluate((e: HTMLVideoElement) => e.currentTime)}s.`);
  return { context, page };
}
