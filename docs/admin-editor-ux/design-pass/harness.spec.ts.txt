async function measure(page: Page) {
  return page.evaluate(() => {
    const vw = document.documentElement.clientWidth;
    const roots = [...document.querySelectorAll('.admin-editor, [role=dialog], main')];
    const visible = (e: Element) => {
      const r = e.getBoundingClientRect(), s = getComputedStyle(e);
      if(!(r.width>0&&r.height>0&&s.visibility!=='hidden'))return false;for(let p:Element|null=e;p;p=p.parentElement){const st=getComputedStyle(p);if(st.clipPath!=='none'||(st.clip!=='auto'&&st.clip!==''))return false;}return true;
    };
    const all = [...new Set(roots.flatMap(r => [r, ...r.querySelectorAll('*')]))].filter(visible);
    const rect = (e: Element) => { const r = e.getBoundingClientRect(); return { x: r.x, y: r.y, width: r.width, height: r.height, right: r.right, bottom: r.bottom }; };
    const name = (e: Element) => e.getAttribute('aria-label') ?? e.textContent?.trim().slice(0, 100) ?? '';
    const outside = all.filter(e => {
      const r = e.getBoundingClientRect();
      if (r.right <= vw + 0.5) return false;
      for (let p = e.parentElement; p; p = p.parentElement) {
        if (['auto', 'scroll'].includes(getComputedStyle(p).overflowX)) return false;
      }
      return true;
    }).map(e => ({ tag: e.tagName, class: String(e.className).slice(0, 90), text: name(e), ...rect(e) }));
    const targets = all.filter(e => e.matches('.admin-lesson-summary button, .admin-lesson-summary a, button[data-move], .transcript-editor-remove, .admin-editor-save button, [role=dialog] button, .document-library > button'))
      .map(e => ({ name: name(e), ...rect(e), min44: e.getBoundingClientRect().width >= 44 && e.getBoundingClientRect().height >= 44 }));
    // Selected new text and comparison text; measurements do not infer glyph ink clipping.
    const texts = all.filter(e => e.matches('.admin-video-state, .admin-video-filename, .transcript-editor-time .ui-description, .admin-editor-save output, .admin-save-hint, .lesson-section-heading p, .admin-lesson-summary a, .admin-lesson-summary button, .document-library > button, [role=dialog] p, [role=dialog] button, main h1'));
    const clipping = texts.map(e => ({ text: name(e), tag: e.tagName, scrollWidth: e.scrollWidth, clientWidth: e.clientWidth, scrollHeight: e.scrollHeight, clientHeight: e.clientHeight, clippedX: e.scrollWidth > e.clientWidth, clippedY:e.scrollHeight>e.clientHeight, ...rect(e) }));
    const rgb = (s: string) => {
      const m = s.match(/^rgba?\(([^)]+)\)$/);
      if (!m) return null;
      const v = m[1].split(/[ ,/]+/).filter(Boolean).map(Number);
      return v.length >= 3 && v.every(Number.isFinite) ? [v[0], v[1], v[2], v[3] ?? 1] : null;
    };
    const lum = (v: number[]) => v.slice(0,3).map(c => c / 255).map(c => c <= .04045 ? c / 12.92 : ((c + .055) / 1.055) ** 2.4).reduce((n,c,i) => n + c * [.2126,.7152,.0722][i], 0);
    const contrast = texts.map(e => {
      const cs = getComputedStyle(e), fg = rgb(cs.color);
      let bg: number[] | null = null, bgCSS = '', reason = '', backgroundElement = '';
      const er = e.getBoundingClientRect();
      const textRects: DOMRect[] = [];
      const walker = document.createTreeWalker(e, NodeFilter.SHOW_TEXT);
      for (let node = walker.nextNode(); node; node = walker.nextNode()) {
        if (!node.textContent?.trim()) continue;
        const range = document.createRange(); range.selectNodeContents(node);
        textRects.push(...Array.from(range.getClientRects()).filter(r => r.width > 0 && r.height > 0));
      }
      if (!textRects.length) reason = 'no rendered text rectangles';
      for (let p: Element | null = e; p; p = p.parentElement) {
        const s = getComputedStyle(p);
        if (Number(s.opacity) !== 1 || s.filter !== 'none' || s.backdropFilter !== 'none' || s.mixBlendMode !== 'normal') { reason = 'ancestor opacity/filter/compositing unsupported'; break; }
        if (!bg) {
          if (s.backgroundImage !== 'none') { reason = 'background image/gradient unsupported'; break; }
          const b = rgb(s.backgroundColor);
          if (!b) { reason = `unsupported background colour ${s.backgroundColor}`; break; }
          if (b[3] > 0 && b[3] < 1) { reason = 'translucent background unsupported'; break; }
          if (b[3] === 1) {
            const r = p.getBoundingClientRect();
            if (textRects.some(t => t.left < r.left - .5 || t.right > r.right + .5 || t.top < r.top - .5 || t.bottom > r.bottom + .5)) { reason = 'text outside solid background bounds'; break; }
            bg = b; bgCSS = s.backgroundColor; backgroundElement = `${p.tagName}.${p.className}`;
          }
        }
      }
      const popover = document.querySelector('.lesson-player-settings');
      if (popover && !popover.contains(e)) { const pr = popover.getBoundingClientRect(); if (textRects.some(t => t.left < pr.right && t.right > pr.left && t.top < pr.bottom && t.bottom > pr.top)) reason ||= 'text box overlaps open popover (paint obscured)'; }
      if (!fg || fg[3] !== 1) reason ||= 'foreground not opaque sRGB';
      if (!bg) reason ||= 'no opaque solid background found';
      const ratio = !reason && fg && bg ? (Math.max(lum(fg),lum(bg))+.05)/(Math.min(lum(fg),lum(bg))+.05) : null;
      return { text: name(e), foreground: cs.color, background: bgCSS, backgroundElement, fontSize: cs.fontSize, textRects: textRects.map(t => ({x:t.x,y:t.y,width:t.width,height:t.height})), ratio, status: ratio === null ? 'NOT_TESTED' : ratio >= 4.5 ? 'PASS' : 'FAIL', reason };
    });
    const panels = [...document.querySelectorAll('.course-editor-panel')].map(e=>({id:e.id,inert:e.hasAttribute('inert'),display:getComputedStyle(e).display,visible:visible(e),...rect(e)}));
    return { viewport: { width: innerWidth, height: innerHeight, clientWidth: vw }, htmlFontSize: getComputedStyle(document.documentElement).fontSize,
      documentScrollWidth: document.documentElement.scrollWidth, overflowPASS: document.documentElement.scrollWidth <= vw && outside.length === 0, outside,
      roots: roots.map(e => ({selector:`${e.tagName}.${e.className}`,...rect(e)})), targets, clipping, contrast, panels,
      displayedPanelCount: panels.filter(p=>p.visible).length,
      theme: { dataTheme: document.documentElement.getAttribute('data-theme'), colorScheme: getComputedStyle(document.documentElement).colorScheme },
    };
  });
}
async function capture(page:Page,state:string,vp:typeof viewports[number]) {
  await page.evaluate(()=>document.fonts.ready);
  await page.mouse.move(0,0);
  await page.waitForTimeout(200);
  const file=`${state}-${vp.name}.jpg`;
  const measured=await measure(page);
  await page.screenshot({path:path.join(evidence,file),type:'jpeg',quality:20,fullPage:state!=='D5'&&state!=='K-move-controls'});
  measurements.push({state,viewportName:vp.name,file,route:new URL(page.url()).pathname,...measured});
  await flush();
  await log(`FACT: Captured ${state} at ${vp.width}×${vp.height} text ×${vp.text}; screenshot design-pass/${file}; measurement index ${measurements.length-1}.`);
}
async function focusState(page:Page){return page.evaluate(()=>{
  const e=document.activeElement; if(!(e instanceof HTMLElement))return null;
  const r=e.getBoundingClientRect(),b=document.querySelector('.admin-editor-save')?.getBoundingClientRect();
  const s=getComputedStyle(e);
  return {tag:e.tagName,name:e.getAttribute('aria-label')??e.getAttribute('name')??e.textContent?.trim().slice(0,100),id:e.id,move:e.getAttribute('data-move'),row:e.closest('[data-row-id]')?.getAttribute('data-row-id'),insideDialog:!!e.closest('[role=dialog]'),focusVisible:e.matches(':focus-visible'),outline:s.outline,rect:{x:r.x,y:r.y,width:r.width,height:r.height,bottom:r.bottom},bar:b?{top:b.top,bottom:b.bottom,left:b.left,right:b.right}:null,covered:!!b&&r.top<b.bottom&&r.bottom>b.top&&r.left<b.right&&r.right>b.left&&!e.closest('.admin-editor-save')};
});}
async function sticky(page:Page,state:string,vp:typeof viewports[number]){
  if(!['390-text200','320'].includes(vp.name))return;
  // Start at the last visible form input, then use actual Tab presses through end controls.
  const fields=page.locator('.admin-editor-form input:visible, .admin-editor-form textarea:visible');
  const n=await fields.count(); if(!n)return;
  await fields.nth(Math.max(0,n-3)).focus();
  const steps=[];
  for(let i=0;i<16;i++){
    const f=await focusState(page);steps.push(f);
    if(await page.locator('.admin-editor-save button').evaluate(e=>e===document.activeElement))break;
    await page.keyboard.press('Tab');
  }
  const file=`${state}-${vp.name}-focus.jpg`;
  await page.screenshot({path:path.join(evidence,file),type:'jpeg',quality:20});
  keyboard.push({check:'sticky-last-fields',state,viewport:vp.name,steps,file,status:steps.some(s=>s?.covered)?'FAIL':'PASS'});
  await flush();await log(`FACT: Tabbed last form fields for ${state}/${vp.name}; focus rect bottoms versus bar top stored in measurements.json keyboard; screenshot ${file}.`);
}
